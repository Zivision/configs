****  General Variables
#+begin_src clojure
;; Get the root directory of the true file (not symlink)
(def project-dir
  (->> *file*
       (java.io.File.)
       (.getCanonicalFile)
       (.getParent)
       (java.io.File.)
       (.getParent)))

(def data
  (->> "/bin/directories.edn"
       (str project-dir)
       (slurp)
       (edn/read-string)))

(def dirs
  (map #(str (fs/home) (:dir %)) data))

(def packages (map #(:name %) data))
#+end_src
****  Utility Functions
#+begin_src clojure
(defn configure-directories [dirs]
  (doseq [d dirs]
    (when-not (= d (str (fs/home) "/"))
      (fs/create-dirs d))))

(defn ensure-hardware-symlink [link target]
  (when-not (fs/sym-link? link)
    (fs/create-sym-link
     link
     (fs/relativize (fs/parent link) target))))

(defn configure-stow-links [packages dirs]
  (doseq [[package d] (map vector packages dirs)]
    (-> (process ["stow" package "-t" d])
        check)))
#+end_src

****  System Config
#+begin_src clojure
;; Config system
(defn sys-config
  "Sets the system up correctly"
  []

  ;; Take the hardware-configuration.nix from /etc/ that's auto generated
  ;; When installing nixos
  (ensure-hardware-symlink
   (str project-dir "/nixos/hardware-configuration.nix")
   "/etc/nixos/hardware-configuration.nix")

  ;; Setup directories
  (configure-directories dirs)

  ;; Use stow to install all config file symlinks
  (configure-stow-links packages dirs))
#+end_src

****  System Rebuild
#+begin_src clojure
;; Rebuild system
(defn sys-rebuild []
  (shell
   (str
    "sudo nixos-rebuild switch --flake \""
    project-dir
    "/nixos\" --impure")))
#+end_src

****  System Update
#+begin_src clojure
;; Update then rebuild
(defn sys-update []
  (shell
   (str
    "nix flake update --flake \""
    project-dir
    "/nixos\""))
  (sys-rebuild))
#+end_src

(defn sys-help []

                 "'-c': Run GNU stow for symlink management along with other commands\n"
                 "'-r': Rebuild NixOS from configs/nixos\n"
                 "'-u': Updates Nixos from configs/nixos\n"
                 "'-h': Shows this page\n"])))

(when (seq *command-line-args*)
  (doseq [arg *command-line-args*]
    (case arg
      "" (sys-help)

      "-r" (sys-rebuild)
      "-u" (sys-update)
      "-h" (sys-help)
