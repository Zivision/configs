***  Get Script Location and Project Root
This finds the original script location
#+begin_src bash
PROJECT_DIR=$(dirname "$(dirname "$(readlink -f "$0")")")
#+end_src
***  Create directories (if needed)

***  Symlink hardware-configuration.nix
#+begin_src bash
if [ ! -L "${PROJECT_DIR}/nixos/hardware-configuration.nix" ]; then
    (cd "${PROJECT_DIR}/nixos" && ln -sr /etc/nixos/hardware-configuration.nix)
fi
#+end_src

***  Run Stow
This will put everything where it needs to go
#+begin_src bash
# Stow packages
(cd "$PROJECT_DIR" && stow doom -t ~/.config/doom)
(cd "$PROJECT_DIR" && stow home -t ~)
(cd "$PROJECT_DIR" && stow bin -t ~/.local/bin)
(cd "$PROJECT_DIR" && stow waybar -t ~/.config/waybar)
(cd "$PROJECT_DIR" && stow hypr -t ~/.config/hypr)
#+end_src
