***  Get Script Location and Project Root
This finds the orignal script location
#+begin_src bash
#!/usr/bin/env sh
PROJECT_DIR=$(dirname "$(dirname "$(readlink -f "$0")")")
#+end_src

***  Run Stow
This will put everything where it needs to go
#+begin_src bash
# Stow packages
(cd "$PROJECT_DIR" && stow doom -t ~/.config/doom)
(cd "$PROJECT_DIR" && stow home -t ~)
(cd "$PROJECT_DIR" && stow bin -t ~/.local/bin)
(cd "$PROJECT_DIR" && sudo stow nixos -t /etc/nixos)
#+end_src
